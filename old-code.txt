function getGateGeometry(type, x, y, w, h) {
  const cx = x + w / 2;
  const cy = y + h * 0.62;
  const gw = Math.min(50, w - 24);
  const gh = Math.min(30, h - 20);
  const left = cx - gw / 2;
  const right = cx + gw / 2;
  const top = cy - gh / 2;
  const bottom = cy + gh / 2;

  let inPins = [];
  let outPin = { x: right + 5, y: cy };

  if (type === 'input') {
    inPins = [];
    outPin = { x: right + 2, y: cy };
  } else if (type === 'output') {
    inPins = [{ x: left - 2, y: cy }];
    outPin = { x: right + 6, y: cy };
  } else if (type === 'not') {
    inPins = [{ x: left - 8, y: cy }];
    outPin = { x: right + 8, y: cy };
  } else if (type === 'and' || type === 'nand') {
    inPins = [
      { x: left - 8, y: cy - gh * 0.24 },
      { x: left - 8, y: cy + gh * 0.24 }
    ];
    outPin = { x: right + (type === 'nand' ? 6 : 3), y: cy };
  } else {
    inPins = [
      { x: left - 10, y: cy - gh * 0.24 },
      { x: left - 10, y: cy + gh * 0.24 }
    ];
    outPin = { x: right + ((type === 'nor' || type === 'xnor') ? 7 : 3), y: cy };
  }

  return { cx, cy, gw, gh, left, right, top, bottom, inPins, outPin };
}

function drawGateShape(type, x, y, w, h, value = false) {
  const geom = getGateGeometry(type, x, y, w, h);
  const { cx, cy, gw, gh, left, right, top, bottom, inPins, outPin } = geom;

  ctx.save();
  ctx.strokeStyle = cssVar('--symbol-stroke');
  ctx.lineWidth = 1.8;
  ctx.lineCap = 'round';
  ctx.lineJoin = 'round';

  inPins.forEach(pin => {
    ctx.beginPath();
    ctx.moveTo(pin.x, pin.y);
    ctx.lineTo(pin.x + 7, pin.y);
    ctx.stroke();
  });

  ctx.beginPath();
  ctx.moveTo(outPin.x - 7, outPin.y);
  ctx.lineTo(outPin.x, outPin.y);
  ctx.stroke();

  if (type === 'not') {
    ctx.beginPath();
    ctx.moveTo(left, top);
    ctx.lineTo(left, bottom);
    ctx.lineTo(right - 8, cy);
    ctx.closePath();
    ctx.stroke();

    ctx.beginPath();
    ctx.arc(right - 3, cy, 3.2, 0, Math.PI * 2);
    ctx.stroke();
    ctx.restore();
    return;
  }

  if (type === 'and' || type === 'nand') {
    const arcCx = left + gw * 0.55;
    const arcR = gh / 2;
    ctx.beginPath();
    ctx.moveTo(left, top);
    ctx.lineTo(left, bottom);
    ctx.lineTo(arcCx, bottom);
    ctx.arc(arcCx, cy, arcR, Math.PI / 2, -Math.PI / 2, true);
    ctx.lineTo(left, top);
    ctx.stroke();

    if (type === 'nand') {
      ctx.beginPath();
      ctx.arc(right - 1.5, cy, 3.1, 0, Math.PI * 2);
      ctx.stroke();
    }
    ctx.restore();
    return;
  }

  if (type === 'or' || type === 'nor' || type === 'xor' || type === 'xnor') {
    const frontX = right - 3;
    const backX = left + 2;
    const backCurveX = left - 7;

    if (type === 'xor' || type === 'xnor') {
      ctx.beginPath();
      ctx.moveTo(backX - 5, top);
      ctx.quadraticCurveTo(backCurveX - 4, cy, backX - 5, bottom);
      ctx.stroke();
    }

    ctx.beginPath();
    ctx.moveTo(backX, top);
    ctx.quadraticCurveTo(backCurveX, cy, backX, bottom);
    ctx.quadraticCurveTo(cx + 6, bottom, frontX, cy);
    ctx.quadraticCurveTo(cx + 6, top, backX, top);
    ctx.stroke();

    if (type === 'nor' || type === 'xnor') {
      ctx.beginPath();
      ctx.arc(right + 1.5, cy, 3.1, 0, Math.PI * 2);
      ctx.stroke();
    }
    ctx.restore();
    return;
  }

  if (type === 'input') {
    const btnR = Math.min(13, gh * 0.48);

    ctx.beginPath();
    ctx.moveTo(left - 2, cy);
    ctx.lineTo(cx - btnR - 4, cy);
    ctx.stroke();

    ctx.fillStyle = value ? cssVar('--accent') : cssVar('--input-off');
    ctx.beginPath();
    ctx.arc(cx - 1, cy, btnR, 0, Math.PI * 2);
    ctx.fill();
    ctx.stroke();

    ctx.lineWidth = 1;
    ctx.strokeStyle = value ? cssVar('--accent') : cssVar('--button-ring-off');
    ctx.beginPath();
    ctx.arc(cx - 1, cy, btnR + 2.2, 0, Math.PI * 2);
    ctx.stroke();

    ctx.lineWidth = 1.8;
    ctx.strokeStyle = cssVar('--symbol-stroke');
    ctx.beginPath();
    ctx.moveTo(cx + btnR + 2, cy);
    ctx.lineTo(right + 1, cy);
    ctx.stroke();

    ctx.restore();
    return;
  }

  if (type === 'output') {
    const bulbR = Math.min(16, gh * 0.78);
    const bulbCx = cx;
    const bulbCy = cy - 1;

    ctx.beginPath();
    ctx.moveTo(left - 2, cy);
    ctx.lineTo(bulbCx - bulbR - 2, cy);
    ctx.stroke();

    ctx.fillStyle = value ? 'rgba(255, 226, 120, 0.42)' : 'rgba(255, 255, 255, 0.03)';
    ctx.beginPath();
    ctx.arc(bulbCx, bulbCy, bulbR, 0, Math.PI * 2);
    ctx.fill();
    ctx.stroke();

    if (value) {
      ctx.fillStyle = 'rgba(255, 214, 80, 0.20)';
      ctx.beginPath(); ctx.arc(bulbCx, bulbCy, bulbR + 7, 0, Math.PI * 2); ctx.fill();
      ctx.fillStyle = 'rgba(255, 240, 170, 0.22)';
      ctx.beginPath(); ctx.arc(bulbCx, bulbCy, bulbR + 3, 0, Math.PI * 2); ctx.fill();
    }

    ctx.lineWidth = 1.2;
    ctx.strokeStyle = value ? '#ffe07a' : cssVar('--bulb-filament-off');
    ctx.beginPath();
    ctx.moveTo(bulbCx - 4, bulbCy + 2);
    ctx.lineTo(bulbCx - 1.5, bulbCy - 1);
    ctx.lineTo(bulbCx + 1.5, bulbCy - 1);
    ctx.lineTo(bulbCx + 4, bulbCy + 2);
    ctx.stroke();

    ctx.lineWidth = 1.8;
    ctx.strokeStyle = cssVar('--symbol-stroke');
    ctx.beginPath();
    ctx.moveTo(bulbCx - 4, bulbCy + bulbR + 1);
    ctx.lineTo(bulbCx + 4, bulbCy + bulbR + 1);
    ctx.stroke();

    if (value) {
      ctx.strokeStyle = 'rgba(255, 214, 80, 0.85)';
      ctx.lineWidth = 1;
      const rays = [
        [0, -(bulbR + 3), 0, -(bulbR + 10)],
        [bulbR + 2, -2, bulbR + 8, -5],
        [-(bulbR + 2), -2, -(bulbR + 8), -5],
        [bulbR - 1, 6, bulbR + 6, 10],
        [-(bulbR - 1), 6, -(bulbR + 6), 10]
      ];
      rays.forEach(([x1, y1, x2, y2]) => {
        ctx.beginPath();
        ctx.moveTo(bulbCx + x1, bulbCy + y1);
        ctx.lineTo(bulbCx + x2, bulbCy + y2);
        ctx.stroke();
      });
    }

    ctx.beginPath();
    ctx.moveTo(bulbCx + bulbR + 2, cy);
    ctx.lineTo(right + 2, cy);
    ctx.stroke();

    ctx.restore();
    return;
  }

  ctx.restore();
}

class Gate {
  constructor(type, x, y) {
    const config = getPortConfig(type);
    this.type = type;
    this.id = `${type}_${nextId++}`;
    this.label = type.toUpperCase();
    this.x = x; this.y = y;
    this.w = 90; this.h = config.height;
    this.value = false;
    this.inputs = [];
    this.inPorts = config.inPorts;
    this.outPorts = config.outPorts;
  }

  getInPos(i) {
    const geom = getGateGeometry(this.type, this.x, this.y, this.w, this.h);
    if (!geom.inPins.length) return { x: this.x, y: this.y + this.h / 2 };
    return geom.inPins[Math.min(i, geom.inPins.length - 1)];
  }

  getOutPos() {
    const geom = getGateGeometry(this.type, this.x, this.y, this.w, this.h);
    return geom.outPin;
  }

  getDeleteBadgePos() {
    return { x: this.x + this.w + 8, y: this.y - 8, r: 9 };
  }

  draw() {
    if (selectedComp === this) {
      ctx.strokeStyle = cssVar('--selection-stroke'); ctx.lineWidth = 1;
      ctx.strokeRect(this.x - 4, this.y - 4, this.w + 8, this.h + 8);

      const del = this.getDeleteBadgePos();
      ctx.fillStyle = '#e74c3c';
      ctx.strokeStyle = '#ffd7d2';
      ctx.lineWidth = 1;
      ctx.beginPath(); ctx.arc(del.x, del.y, del.r, 0, Math.PI * 2); ctx.fill(); ctx.stroke();

      ctx.strokeStyle = '#ffffff';
      ctx.lineWidth = 1.8;
      ctx.beginPath();
      ctx.moveTo(del.x - 3.2, del.y - 3.2);
      ctx.lineTo(del.x + 3.2, del.y + 3.2);
      ctx.moveTo(del.x + 3.2, del.y - 3.2);
      ctx.lineTo(del.x - 3.2, del.y + 3.2);
      ctx.stroke();
    }

    ctx.fillStyle = cssVar('--symbol-label'); ctx.font = "10px Inter"; ctx.textAlign = "center";
    ctx.fillText(this.label, this.x + this.w/2, this.y + 12);

    drawGateShape(this.type, this.x, this.y, this.w, this.h, this.value);

    ctx.lineWidth = 1;
    for(let i=0; i<this.inPorts; i++) {
      let p = this.getInPos(i);
      ctx.fillStyle = (connectingFrom && connectingFrom.component === this && connectingFrom.portIndex === i) ? cssVar('--port-connect-highlight') : (this.inputs[i] ? cssVar('--accent') : cssVar('--port-off'));
      ctx.strokeStyle = cssVar('--port-ring');
      ctx.beginPath(); ctx.arc(p.x, p.y, 4, 0, Math.PI*2); ctx.fill(); ctx.stroke();
    }
    if(this.outPorts) {
      let p = this.getOutPos();
      ctx.fillStyle = this.value ? cssVar('--accent') : cssVar('--port-off');
      ctx.strokeStyle = cssVar('--port-ring');
      ctx.beginPath(); ctx.arc(p.x, p.y, 4, 0, Math.PI*2); ctx.fill(); ctx.stroke();
    }
  }

  eval() {
    if(this.type==='input') return;
    let a=this.inputs[0]||false, b=this.inputs[1]||false;
    switch(this.type) {
      case 'and': this.value = a&&b; break;
      case 'or': this.value = a||b; break;
      case 'not': this.value = !a; break;
      case 'xor': this.value = a!==b; break;
      case 'xnor': this.value = a===b; break;
      case 'nand': this.value = !(a&&b); break;
      case 'nor': this.value = !(a||b); break;
      case 'output': this.value = a; break;
    }
  }
}

function validateWire(wire) {
  if (!wire.to || !wire.to.c || wire.to.c.type === 'input') return false;
  if (wire.to.p >= wire.to.c.inPorts) return false;
  if (!wire.from || !wire.from.c || !wire.from.c.outPorts) return false;
  return true;
}

function getErrorWires() {
  return wires.filter(w => !validateWire(w));
}

function serializeCircuit() {
  const data = {
    components: components.map(c => ({
      type: c.type,
      x: c.x,
      y: c.y,
      label: c.label,
      value: c.value
    })),
    wires: wires.map(w => {
      const fromIdx = components.indexOf(w.from.c);
      const toIdx = components.indexOf(w.to.c);
      return { from: fromIdx, fromPort: w.from.p, to: toIdx, toPort: w.to.p };
    })
  };
  return btoa(JSON.stringify(data));
}

function deserializeCircuit(encoded) {
  try {
    const data = JSON.parse(atob(encoded));
    clearWorkspace();
    
    data.components.forEach(comp => {
      const gate = new Gate(comp.type, comp.x, comp.y);
      gate.label = comp.label;
      gate.value = comp.value;
      components.push(gate);
    });
    
    data.wires.forEach(w => {
      if (w.from >= 0 && w.to >= 0 && w.from < components.length && w.to < components.length) {
        wires.push({
          from: { c: components[w.from], p: w.fromPort },
          to: { c: components[w.to], p: w.toPort }
        });
      }
    });
    
    selectedComp = null;
    simulate();
    return true;
  } catch (e) {
    console.error('Deserialization failed:', e);
    return false;
  }
}

function shareCircuit() {
  const encoded = serializeCircuit();
  const url = window.location.origin + window.location.pathname + '#circuit=' + encoded;
  prompt('Share this link:', url);
}

function loadFromURL() {
  const hash = window.location.hash.substring(1);
  if (hash.startsWith('circuit=')) {
    const encoded = hash.substring(8);
    if (deserializeCircuit(encoded)) {
      setMode('props');
    }
  }
}

  panY = 0;
}

function createGate(type, x, y, label) {
  const gate = new Gate(type, x, y);
  if (label) gate.label = label;
  components.push(gate);
  return gate;
}

function connect(fromGate, toGate, toPort) {
  wires.push({ from: { c: fromGate, p: 0 }, to: { c: toGate, p: toPort } });
}

function escapeHtml(text) {
  return String(text)
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;');
}

function toggleSelectedInput() {
  if (selectedComp && selectedComp.type === 'input') {
    selectedComp.value = !selectedComp.value;
    pushUndo();
    simulate();
  }
}

function deleteSelectedComponent() {
  if (!selectedComp) return;
  pushUndo();
  components = components.filter(c => c !== selectedComp);
  wires = wires.filter(w => w.from.c !== selectedComp && w.to.c !== selectedComp);
  selectedComp = null;
  simulate();
}

function renderTruthTablePanel() {
  const el = getModeContentEl();
  const ins = components.filter(c => c.type === 'input');
  const outs = components.filter(c => c.type === 'output');

  if (!ins.length || !outs.length) {
    el.innerHTML = `<div class="empty-state"><p>Truth Table requires at least 1 Input and 1 Output component on the canvas.</p></div>`;
    return;
  }

  if (ins.length > 8) {
    el.innerHTML = `<div class="empty-state"><p>Truth Table is limited to 8 or fewer inputs.</p></div>`;
    return;
  }

  const originalValues = ins.map(i => i.value);
  const rowCount = 1 << ins.length;

  let tableHtml = '<div style="overflow-x: auto;"><table id="truth-table-view"><thead><tr>';
  ins.forEach(i => { tableHtml += `<th>${escapeHtml(i.label)}</th>`; });
  outs.forEach(o => { tableHtml += `<th style="color:var(--accent);">${escapeHtml(o.label)}</th>`; });
  tableHtml += '</tr></thead><tbody>';

  let currentPattern = 0;
  ins.forEach((input, idx) => {
    if (input.value) currentPattern |= (1 << (ins.length - 1 - idx));
  });

  for (let r = 0; r < rowCount; r++) {
    const isCurrent = (r === currentPattern);
    tableHtml += `<tr class="${isCurrent ? 'current-row' : ''}">`;

    ins.forEach((input, idx) => {
      const bit = (r >> (ins.length - 1 - idx)) & 1;
      input.value = !!bit;
      tableHtml += `<td class="${bit ? 'active-bit' : ''}">${bit}</td>`;
    });

    for (let k = 0; k < 6; k++) {
      wires.forEach(w => { w.to.c.inputs[w.to.p] = w.from.c.value; });
      components.forEach(c => { if (c.type !== 'input') c.eval(); });
    }

    outs.forEach(out => {
      const bit = out.value ? 1 : 0;
      tableHtml += `<td class="${bit ? 'active-bit' : ''}">${bit}</td>`;
    });

    tableHtml += '</tr>';
  }

  tableHtml += '</tbody></table></div>';

  ins.forEach((input, idx) => { input.value = originalValues[idx]; });
  for (let k = 0; k < 6; k++) {
    wires.forEach(w => { w.to.c.inputs[w.to.p] = w.from.c.value; });
    components.forEach(c => c.eval());
  }

  el.innerHTML = tableHtml;
}

function updateUI() {
  if (counts) {
    const errorCount = getErrorWires().length;
    let text = `Gates: ${components.length} &nbsp;&nbsp; Wires: ${wires.length}`;
    if (errorCount > 0) {
      text += ` &nbsp;&nbsp; <span style="color:var(--wire-error);">Errors: ${errorCount}</span>`;
    }
    counts.innerHTML = text;
  }

  const zoomInd = document.getElementById('zoom-indicator');
  if (zoomInd) {
    zoomInd.textContent = `Zoom: ${Math.round(zoomLevel * 100)}%`;
  }

  updateUndoRedoButtons();

  if (currentMode === 'examples') renderExampleBrowser();
  else if (currentMode === 'challenges') renderChallengeBrowser();
  else if (currentMode === 'props') renderPropertiesPanel();
  else if (currentMode === 'tt') renderTruthTablePanel();
}

function simulate() {
  components.forEach(c => { if (c.type !== 'input') c.inputs = []; });
  for (let k = 0; k < 6; k++) {
    wires.forEach(w => {
      if (w.from.c && w.to.c) {
        w.to.c.inputs[w.to.p] = w.from.c.value;
      }
    });
    components.forEach(c => c.eval());
  }
  render();
  updateUI();
}

function render() {
  const dpr = window.devicePixelRatio || 1;
  ctx.setTransform(1, 0, 0, 1, 0, 0);
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  ctx.save();
  ctx.scale(dpr, dpr);
  ctx.translate(panX, panY);
  ctx.scale(zoomLevel, zoomLevel);

  wires.forEach(w => {
    if (!w.from.c || !w.to.c) return;
    const p1 = w.from.c.getOutPos();
    const p2 = w.to.c.getInPos(w.to.p);

    const isValid = validateWire(w);
    const isActive = w.from.c.value;

    ctx.strokeStyle = !isValid
      ? cssVar('--wire-error')
      : (isActive ? cssVar('--accent') : cssVar('--wire-off'));
    ctx.lineWidth = isActive ? 2.5 : 2;

    const dx = Math.abs(p2.x - p1.x) / 2;
    const cp = Math.max(dx, 30);

    ctx.beginPath();
    ctx.moveTo(p1.x, p1.y);
    ctx.bezierCurveTo(p1.x + cp, p1.y, p2.x - cp, p2.y, p2.x, p2.y);
    ctx.stroke();
  });

  if (connectingFrom) {
    const p1 = connectingFrom.pos;
    const p2 = mouseWorldPos;
    ctx.strokeStyle = cssVar('--wire-preview');
    ctx.lineWidth = 2;
    ctx.setLineDash([4, 4]);
    ctx.beginPath();
    ctx.moveTo(p1.x, p1.y);
    ctx.lineTo(p2.x, p2.y);
    ctx.stroke();
    ctx.setLineDash([]);
  }

  components.forEach(c => c.draw());

  ctx.restore();
}

// Transforms screen mouse/touch coordinates into world canvas coordinates
function getEventWorldPos(e) {
  const rect = canvas.getBoundingClientRect();
  const clientX = (e.touches && e.touches.length > 0) ? e.touches[0].clientX : e.clientX;
  const clientY = (e.touches && e.touches.length > 0) ? e.touches[0].clientY : e.clientY;
  const screenX = clientX - rect.left;
  const screenY = clientY - rect.top;
  return {
    x: (screenX - panX) / zoomLevel,
    y: (screenY - panY) / zoomLevel,
    screenX,
    screenY
  };
}

function getComponentAt(wx, wy) {
  for (let i = components.length - 1; i >= 0; i--) {
    const c = components[i];
    if (wx >= c.x - 4 && wx <= c.x + c.w + 4 && wy >= c.y - 4 && wy <= c.y + c.h + 4) {
      return c;
    }
  }
  return null;
}

function getPortAt(wx, wy) {
  const radius = 12 / zoomLevel;

  for (let i = components.length - 1; i >= 0; i--) {
    const c = components[i];

    if (c.outPorts) {
      const p = c.getOutPos();
      if (Math.hypot(wx - p.x, wy - p.y) <= radius) {
        return { component: c, type: 'out', portIndex: 0, pos: p };
      }
    }

    for (let j = 0; j < c.inPorts; j++) {
      const p = c.getInPos(j);
      if (Math.hypot(wx - p.x, wy - p.y) <= radius) {
        return { component: c, type: 'in', portIndex: j, pos: p };
      }
    }
  }
  return null;
}

function handlePointerDown(e) {
  if (e.button !== undefined && e.button !== 0) return;

  const pos = getEventWorldPos(e);
  mouseWorldPos = { x: pos.x, y: pos.y };
  startPointerPos = { x: pos.screenX, y: pos.screenY };
  isPointerDown = true;
  hasDragged = false;

  // 1. Check if delete badge was clicked on selected component
  if (selectedComp) {
    const del = selectedComp.getDeleteBadgePos();
    if (Math.hypot(pos.x - del.x, pos.y - del.y) <= del.r + 4) {
      deleteSelectedComponent();
      isPointerDown = false;
      return;
    }
  }

  // 2. Check port click for wiring
  const port = getPortAt(pos.x, pos.y);
  if (port) {
    connectingFrom = port;
    render();
    return;
  }

  // 3. Check component click
  const comp = getComponentAt(pos.x, pos.y);
  if (comp) {
    selectedComp = comp;
    draggedComp = comp;
    dragOffX = pos.x - comp.x;
    dragOffY = pos.y - comp.y;
    updateUI();
    render();
    return;
  }

  // 4. Clicked empty background -> Deselect and enable canvas panning
  selectedComp = null;
  connectingFrom = null;
  isPanning = true;
  panStartX = pos.screenX - panX;
  panStartY = pos.screenY - panY;
  updateUI();
  render();
}

function handlePointerMove(e) {
  const pos = getEventWorldPos(e);
  mouseWorldPos = { x: pos.x, y: pos.y };

  if (isPointerDown) {
    const dist = Math.hypot(pos.screenX - startPointerPos.x, pos.screenY - startPointerPos.y);
    if (dist > 4) {
      hasDragged = true;
    }
  }

  if (draggedComp) {
    draggedComp.x = pos.x - dragOffX;
    draggedComp.y = pos.y - dragOffY;
    render();
  } else if (isPanning) {
    panX = pos.screenX - panStartX;
    panY = pos.screenY - panStartY;
    render();
  } else if (connectingFrom) {
    render();
  }
}

function handlePointerUp(e) {
  if (!isPointerDown) return;

  const pos = getEventWorldPos(e);

  if (connectingFrom) {
    const targetPort = getPortAt(pos.x, pos.y);
    if (targetPort && targetPort.component !== connectingFrom.component) {
      let fromPort = null, toPort = null;
      if (connectingFrom.type === 'out' && targetPort.type === 'in') {
        fromPort = connectingFrom;
        toPort = targetPort;
      } else if (connectingFrom.type === 'in' && targetPort.type === 'out') {
        fromPort = targetPort;
        toPort = connectingFrom;
      }

      if (fromPort && toPort) {
        wires = wires.filter(w => !(w.to.c === toPort.component && w.to.p === toPort.portIndex));
        wires.push({
          from: { c: fromPort.component, p: fromPort.portIndex },
          to: { c: toPort.component, p: toPort.portIndex }
        });
        pushUndo();
        simulate();
      }
    }
    connectingFrom = null;
    render();
  } else if (draggedComp) {
    if (!hasDragged && draggedComp.type === 'input') {
      // Direct click on input component -> Toggle Logic State (0 <-> 1)
      draggedComp.value = !draggedComp.value;
      pushUndo();
      simulate();
    } else if (hasDragged) {
      pushUndo();
    }
    draggedComp = null;
  }

  isPanning = false;
  isPointerDown = false;
  hasDragged = false;
}
